/* ------------------------------------------------------------------
   4. TRADE IDEA LOG (FULL ANALYSIS)
   ------------------------------------------------------------------ */
CREATE TABLE ict_trade_ideas (
    idea_id           SERIAL PRIMARY KEY,
    asset             VARCHAR(10) NOT NULL,
    bias_htf          VARCHAR(6),   -- Bullish / Bearish
    bias_ltf          VARCHAR(6),   -- Entry‑timeframe bias
    ob_id             INT REFERENCES ict_definitions(concept_id),    -- Order Block
    fvg_id            INT REFERENCES ict_definitions(concept_id),    -- Fair Value Gap
    liquidity_id      INT REFERENCES ict_definitions(concept_id),    -- Liquidity Pool / Void
    pd_array_id       INT REFERENCES ict_definitions(concept_id),    -- Discount / Premium
    model_id          INT REFERENCES ict_models(model_id),
    sweep_id          INT REFERENCES ict_definitions(concept_id),    -- Sweep / Void
    window_id         INT REFERENCES ict_time_windows(window_id),
    confluence_count  SMALLINT,     -- 0‑6 confirmations
    star_rating       SMALLINT,     -- 1‑5 stars
    stop_level        NUMERIC(12,6),
    tp_level          NUMERIC(12,6),
    risk_perc         NUMERIC(5,2), -- % of account risked
    entry_time        TIMESTAMP,
    exit_time         TIMESTAMP,
    pnl_percent       NUMERIC(8,2), -- % PnL
    notes             TEXT
);

/* ------------------------------------------------------------------
   5. PERFORMANCE METRICS (computed view)
   ------------------------------------------------------------------ */
CREATE VIEW ict_performance AS
SELECT
    bias_htf,
    bias_ltf,
    model_id,
    COUNT(*)                AS trade_count,
    SUM(CASE WHEN pnl_percent>0 THEN 1 ELSE 0 END)::float / COUNT(*)  AS win_rate,
    AVG(pnl_percent)        AS avg_expectancy,
    SUM(CASE WHEN pnl_percent>0 THEN pnl_percent ELSE 0 END) /
    SUM(CASE WHEN pnl_percent<0 THEN -pnl_percent ELSE 0 END) AS profit_factor,
    STDDEV(pnl_percent)     AS std_dev,
    MIN(pnl_percent)        AS min_pnl,
    MAX(pnl_percent)        AS max_pnl
FROM ict_trade_ideas
GROUP BY bias_htf, bias_ltf, model_id;
